# P5.5 三档产出形态与可感知 Blender 的讨论模式 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让低模链路支持「道具 / 盒式舞台 / 开放场景」三档产出形态，并让讨论模式能只读地查看 Blender 场景与对象参数。

**Architecture:** 新增 `ShapePolicy`（档位策略表）作为唯一分档来源，摘要模板、规划器提示词、执行纪律、自检截图策略、步数预算全部按档位读表，通用规则保持一份。讨论模式从「单次流式、工具为空」升级为独立的 `DiscussionAgent`：工具白名单只给三个只读语义（`get_scene_info` / `look` / 自造的 `inspect_object`），载入资产 .blend 前后端走 `sceneLoad.*` 往返弹窗（无超时），未连接或忙碌则静默降级为纯文字讨论。

**Tech Stack:** C++20 / Qt 6.10.3（Core、WebSockets、Test）、Qt Test + MockLlmServer + MockMcpHarness、Vue 3 + TypeScript + Vitest + jsdom、CMake（`qt_add_resources` 显式文件列表）。

```cpp
#pragma once

#include <QString>

// 低模产出形态三档。写死「正方形微缩场景」的根因在三层提示词里，
// 分档后每档有自己的摘要模板、规划器提示词、执行纪律、自检方式与步数预算。
enum class OutputShape
{
    Prop,      // 道具：独立单体资产（或一组可分离小件），无场景布局，可复用
    Diorama,   // 盒式舞台：方形底盘约束的舞台场景，构图限定在方形区域内
    OpenScene, // 开放场景：无方形边界的大空间，支持长轨道、跨区域与时序动画
};

OutputShape shapeFromString(const QString& text); // 未知/空 → Diorama
QString shapeKey(OutputShape shape);              // 落盘用键名
QString shapeLabel(OutputShape shape);            // 界面与提示词用中文名

// 每档的可变参数。纪律与提示词正文用自由函数返回（长文本，塞进聚合初始化没法读）。
struct ShapePolicy
{
    QString summarizeTemplate;
    QString askTemplate;
    int budgetPerGroup = 30;
    int budgetCap = 200;
    int minGroups = 1;
    int maxGroups = 6;
};

const ShapePolicy& shapePolicy(OutputShape shape);
int shapeBudget(OutputShape shape, int groupCount);

QString planSystemPromptFor(OutputShape shape);
QString shapeDiscipline(OutputShape shape);  // 接在通用布局纪律之后
QString shapeSelfCheck(OutputShape shape);   // 截图几次、从哪些机位
```

```cpp
#include "core/agent/ShapePolicy.h"

namespace {

const char* kPropKey = "prop";
const char* kDioramaKey = "diorama";
const char* kOpenKey = "open";

// 三份规划器提示词共用一段「JSON 规范」说明，只在组数与施工顺序上分叉：
// 复制三整份以后必然改一处漏两处。
QString planJsonSpec(int minGroups, int maxGroups, const QString& order)
{
    return QStringLiteral(
               "你是 SynthForge Studio 的低模建造规划器。只输出一个 JSON 对象，"
               "不要任何解释或代码块围栏，格式："
               "{\"rules\":[\"本任务特有的跨组约束\"],\"groups\":[{\"title\":\"分组名\","
               "\"items\":[\"物体1\"],"
               "\"placements\":[{\"name\":\"物体1\",\"where\":\"相对位置与朝向\"}],"
               "\"acceptance\":\"自检时截图检查什么\"}]}。"
               "要求：%1~%2 组，覆盖用户提示词中全部必须独立的物体与数量下限；"
               "按施工顺序排列（%3）。"
               "可以在主题内合理补充道具，但不得添加与主题无关的物体，"
               "也不得为凑组数拆出杂物组；留白优先于堆砌。"
               "items 里的每个物体必须在 placements 里有一条同名条目；"
               "where 只写相对关系不写绝对坐标。"
               "rules 只写本任务特有、跨组生效的约束，"
               "不要重复贴地、不重叠、单位这类通用规则。")
        .arg(minGroups)
        .arg(maxGroups)
        .arg(order);
}

} // namespace

OutputShape shapeFromString(const QString& text)
{
    if (text == QLatin1String(kPropKey))
        return OutputShape::Prop;
    if (text == QLatin1String(kOpenKey))
        return OutputShape::OpenScene;
    return OutputShape::Diorama; // 未知值一律回落：ui.ini 被手改坏也不能顶崩界面
}

QString shapeKey(OutputShape shape)
{
    switch (shape) {
    case OutputShape::Prop:
        return QLatin1String(kPropKey);
    case OutputShape::OpenScene:
        return QLatin1String(kOpenKey);
    case OutputShape::Diorama:
        break;
    }
    return QLatin1String(kDioramaKey);
}

QString shapeLabel(OutputShape shape)
{
    switch (shape) {
    case OutputShape::Prop:
        return QStringLiteral("道具");
    case OutputShape::OpenScene:
        return QStringLiteral("开放场景");
    case OutputShape::Diorama:
        break;
    }
    return QStringLiteral("盒式舞台");
}

const ShapePolicy& shapePolicy(OutputShape shape)
{
    static const ShapePolicy kProp{QStringLiteral(":/prompts/summarizer-prop.md"),
                                   QStringLiteral(":/prompts/ask-prop.md"), 20, 40, 1, 2};
    static const ShapePolicy kDiorama{QStringLiteral(":/prompts/summarizer.md"),
                                      QStringLiteral(":/prompts/ask.md"), 30, 200, 3, 6};
    static const ShapePolicy kOpen{QStringLiteral(":/prompts/summarizer-open.md"),
                                   QStringLiteral(":/prompts/ask-open.md"), 40, 320, 3, 8};
    switch (shape) {
    case OutputShape::Prop:
        return kProp;
    case OutputShape::OpenScene:
        return kOpen;
    case OutputShape::Diorama:
        break;
    }
    return kDiorama;
}

int shapeBudget(OutputShape shape, int groupCount)
{
    const ShapePolicy& policy = shapePolicy(shape);
    // 一轮对话算一步、一次工具调用不算一步：道具一次成型，舞台要反复截图修，
    // 开放场景对象多且按机位/按帧多次截图，所以单组预算与总封顶都更高。
    return qMin(policy.budgetPerGroup * qMax(1, groupCount), policy.budgetCap);
}

QString planSystemPromptFor(OutputShape shape)
{
    const ShapePolicy& policy = shapePolicy(shape);
    switch (shape) {
    case OutputShape::Prop:
        return planJsonSpec(policy.minGroups, policy.maxGroups,
                            QStringLiteral("主体 → 附属小件（超过 2 组就是拆得太碎）"));
    case OutputShape::OpenScene:
        return planJsonSpec(policy.minGroups, policy.maxGroups,
                            QStringLiteral(
                                "地面与主轴 → 近景分区 → 中景分区 → 远景分区"
                                "（同一分区的物体放在同一组）"));
    case OutputShape::Diorama:
        break;
    }
    return planJsonSpec(policy.minGroups, policy.maxGroups,
                        QStringLiteral("底座环境 → 建筑外观 → 室内陈设 → 灯光特效"));
}

QString shapeDiscipline(OutputShape shape)
{
    switch (shape) {
    case OutputShape::Prop:
        return QStringLiteral(
            "道具档纪律（本档专用，逐条执行）：\n"
            "P1. 不要任何底座、地面、地平面对象：资产外的空处就是空处。\n"
            "P2. 所有物体挂在一个根空对象下，根对象名 = 资产英文名，"
            "位于世界原点；每个子物体自己的 origin 落在其底部中心。\n"
            "P3. 不做雾、粒子、光雾、飞虫、湿地面等氛围特效。\n"
            "P4. 只允许属于物体本身的功能光源（台灯、屏幕、灯管），"
            "不要太阳光、环境光、世界环境设置。\n"
            "P5. 子物体不得合并成一个网格：进 Unity 后要能单独取用。");
    case OutputShape::OpenScene:
        return QStringLiteral(
            "开放场景档纪律（本档专用，逐条执行）：\n"
            "O1. 不设方形边界：沿规划给出的主轴展开，近景 / 中景 / 远景三段都要有内容，"
            "禁止把物体都堆在原点一圈。\n"
            "O2. 长条结构（轨道、道路、河沿、围墙）按重复段搭建，"
            "总长度必须达到主轴要求，不得为了省事在中途截断。\n"
            "O3. 氛围层（雾、体积光、粒子）照做，但必须放进名为 FX 的集合里"

            "不能因为「反正被雾挡住」就只做到中途，导出后 FX 不存在。\n"
            "O5. 运动物体的位移 / 旋转必须有独立 action（首尾一致以便无缝循环），"
            "不要用 driver 或约束驱动（约束请先烘焙成关键帧）。");
    case OutputShape::Diorama:
        break;
    }
    return QStringLiteral(
        "盒式舞台档纪律（本档专用，逐条执行）：\n"
        "D1. 地面是一个完整的正方形底座，所有元素都落在底座内，不得伸出边缘。\n"
        "D2. 底座边长按真实尺度反推：放得下主体完整立面、入口前至少 2 米人行道"
        "与近景道具；不得为了压缩画面把主体、门窗或道具做小。\n"
        "D3. 构图紧凑、有近 / 中 / 远三层纵深，呈现小巧精致的模型感。\n"
        "D4. 氛围层（雾、体积光、粒子）照做，但必须放进名为 FX 的集合里"
        "（FX 集合不会进入 glb/fbx，只在 .blend 里保留）。");
}

{
    switch (shape) {
```

```cpp
#include <QJsonObject>

namespace {

// Python 字符串字面量：借 JSON 的转义规则，路径里的反斜杠与引号都不用手工处理
QString pyLiteral(const QString& text)
{
    const QJsonArray wrap{text};
    return QString::fromUtf8(QJsonDocument(wrap).toJson(QJsonDocument::Compact))
        .mid(1) // 去掉开头的 '['
        .chopped(1)
        .remove(QRegularExpression(QStringLiteral(",\\s*$")));
}

} // namespace

QString resolveBlendSource(const AssetLibrary& library, const QString& assetPath)
{
    if (assetPath.isEmpty())
        return {};
    const QFileInfo picked(library.absoluteFor(assetPath));
    // 选中项本身是 .blend 就用它；否则取同基名的 .blend（云资产只有 glb，返回空）
    const QString blend = picked.suffix().compare(QLatin1String("blend"), Qt::CaseInsensitive) == 0
                              ? picked.absoluteFilePath()
                              : QDir(picked.dir().absolutePath())
                                    .filePath(picked.completeBaseName() + QStringLiteral(".blend"));
    return QFileInfo::exists(blend) ? blend : QString();
}

QString sceneLoadScript(const QString& targetBlend, const QString& backupDir)
{
    return QStringLiteral(
               "import bpy, json, os, time\n"
               "target = %1\n"
               "backup_dir = %2\n"
               "backup = ''\n"
               "if bpy.data.objects:\n"
               "    os.makedirs(backup_dir, exist_ok=True)\n"
               "    backup = os.path.join(backup_dir, 'current_' + time.strftime('%Y%%m%%d_%%H%%M%%S') + '.blend')\n"
               "    prefs = bpy.context.preferences.filepaths\n"
               "    old_save_version = prefs.save_version\n"
               "    prefs.save_version = 0\n"
               "    try:\n"
               "        bpy.ops.wm.save_as_mainfile(filepath=backup)\n"
               "    finally:\n"
               "        prefs.save_version = old_save_version\n"
               "bpy.ops.wm.open_mainfile(filepath=target)\n"
               "print('LOADED=' + target)\n"
               "print('BACKUP=' + backup)\n")
        .arg(pyLiteral(targetBlend), pyLiteral(QDir(backupDir).absolutePath()));
}
```

```cpp
            QJsonObject{{"area", area},
                        {"requestId", requestId},
                        {"file", QFileInfo(blend).completeBaseName()},
                        {"path", blend}});
        return QJsonValue(QJsonObject{{"accepted", true}, {"requestId", requestId}});
    });

        const QString area = m_sceneLoadRequests.take(params.value("requestId").toString());
        if (!area.isEmpty())
            m_sceneLoadTargets.remove(area);
        return QJsonValue(QJsonObject{{"ok", true}, {"cancelled", true}});
    });
    m_router->registerHandler(QStringLiteral("sceneLoad.accept"), [this](const QJsonObject& params) {
        const QString requestId = params.value(QStringLiteral("requestId")).toString();
        const QString area = m_sceneLoadRequests.take(requestId);
        if (area.isEmpty()) // 旧弹窗或已取消：绝不重复载入
            return QJsonValue(QJsonObject{{"ok", false}, {"error", "request-not-found"}});
        const QString blend = m_sceneLoadTargets.take(area);

            blend, QDir(m_uiState->path()).filePath(QStringLiteral("backups/discussion")));
        m_mcp->callTool(QStringLiteral("execute_blender_code"), QJsonObject{{"code", script}},
                        [this, area, blend](const QJsonValue&, const QJsonObject& error) {

                                    QStringLiteral("sceneLoad.notice"),

                        });
```

```cpp
QString inspectSchema()

QString pyString(const QString& text)
{
    QString out;
    for (const QChar ch : text) {
        switch (ch.unicode()) {
        case u'\\':
            out += QLatin1String("\\\\");
            break;
        case u'\'':
            out += QLatin1String("\\'");
            break;
        case u'\n':
            out += QLatin1String("\\n");
            break;
        case u'\r':
            out += QLatin1String("\\r");
            break;
        case u'\t':
            out += QLatin1String("\\t");
            break;
        default:
            out += ch;
        }
    }
    return out;
}

} // namespace

bool isSafeObjectName(const QString& name)
{
    // 名字会被塞进 Python 字面量：白名单字符之外一律拒（含分号、引号、控制符、空、超长）
    if (name.isEmpty() || name.size() > 64)
        return false;
    static const QRegularExpression pattern(QStringLiteral("^[A-Za-z0-9_.\\-\\u4e00-\\u9fff]+$"));
    return pattern.match(name).hasMatch();
}

QString inspectObjectScript(const QString& name)
{
    return QStringLiteral(
               "import bpy, json\n"
               "name = '%1'\n"
               "ob = bpy.data.objects.get(name)\n"
               "if ob is None:\n"
               "    print(json.dumps({'error': 'object-not-found',\n"
               "                      'candidates': sorted(o.name for o in bpy.data.objects)[:40]},\n"
               "                     ensure_ascii=False))\n"
               "else:\n"
               "    action = None\n"
               "    if ob.animation_data and ob.animation_data.action:\n"
               "        action = ob.animation_data.action.name\n"
               "    info = {\n"
               "        'name': ob.name,\n"
               "        'type': ob.type,\n"
               "        'parent': ob.parent.name if ob.parent else None,\n"
               "        'children': [c.name for c in ob.children][:40],\n"
               "        'location': [round(v, 3) for v in ob.location],\n"
               "        'rotation_euler': [round(v, 3) for v in ob.rotation_euler],\n"
               "        'scale': [round(v, 3) for v in ob.scale],\n"
               "        'dimensions': [round(v, 3) for v in ob.dimensions],\n"
               "        'materials': [m.name for m in ob.data.materials] if hasattr(ob.data, 'materials') else [],\n"
               "        'modifiers': [m.name for m in ob.modifiers],\n"
               "        'action': action,\n"
               "    }\n"
               "    print(json.dumps(info, ensure_ascii=False))\n")
        .arg(pyString(name));
}

QJsonArray discussionTools(const QJsonArray& mcpTools)
{
    QJsonArray allowed = ToolBridge::mcpToolsToOpenAi(mcpTools);
    QJsonArray kept;
    for (const QJsonValue& value : allowed) {

void DiscussionAgent::ask(const QString& area, const QString& systemPrompt,
                          const QJsonArray& history, const QString& userText, bool withTools)
{
    m_area = area;
    m_stream.clear();
    m_visible.clear();
    m_toolCalls = 0;
    m_inFlight = 0;
    m_roundDone = false;
    m_tools = withTools && m_mcp && m_mcp->isReady() ? discussionTools(m_mcpTools) : QJsonArray{};
    m_messages = QJsonArray{QJsonObject{{"role", "system"}, {"content", systemPrompt}}};
    for (const QJsonValue& item : history)
        m_messages.append(item);
    m_messages.append(QJsonObject{{"role", "user"}, {"content", userText}});
    streamOnce();
}

void DiscussionAgent::streamOnce()
{
    auto filter = std::make_shared<AskStreamFilter>();
    m_llm->chatStream(
        m_cfg, m_messages, m_tools,
        {.onDelta =
             [this, filter](const QString& delta) {
                 m_stream += delta;
                 const QString show = filter->feed(delta);
                 if (!show.isEmpty()) {
                     m_visible += show;
                     emit delta(m_area, show);
                 }
             },
         .onToolCalls = [this](const QJsonArray& calls) { executeCalls(calls); },
         .onError = [this](const QString& error) { emit failed(m_area, error); },
         .onDone =
             [this, filter]() {
                 if (m_inFlight > 0 || m_roundDone)
                     return; // 工具在途：等最后一个结果回来再继续；已收尾：不双发
                 const QString tail = filter->finish();
                 if (!tail.isEmpty()) {
                     m_visible += tail;
                     emit delta(m_area, tail);
                 }
                 finishRound();
             }},
        m_area);
}

void DiscussionAgent::finishRound()
{
    if (m_roundDone)
        return;
    m_roundDone = true;
    const AskReply reply = parseAskReply(m_stream);
    // visible 优先给已流式出去的 m_visible：前端气泡只有这些字，落库必须一致，
    // 否则刷新后内容会变长
    emit finished(m_area, m_visible.isEmpty() ? reply.visible : m_visible, reply.executable);
}

void DiscussionAgent::executeCalls(const QJsonArray& calls)
{
    if (m_toolCalls >= kMaxAskToolCalls) {
        // 预算用完：这一轮当作「无工具」收尾，不再回喂，避免无限看图
        const QString note = QStringLiteral("（查看次数已达上限，以下结论基于已看到的内容。）");
        m_visible += note;
        emit delta(m_area, note);
        finishRound();
        return;
    }
    // 回喂历史必须转回 OpenAI wire format（与 AgentOrchestrator 同规则）
    m_messages.append(QJsonObject{{"role", "assistant"},
                                  {"content", m_stream},
                                  {"tool_calls", ToolBridge::internalCallsToWire(calls)}});
    m_inFlight = calls.size();
    for (const QJsonValue& value : calls) {
        const QJsonObject call = value.toObject();
        const QString id = call.value(QStringLiteral("id")).toString();
        const QString name = call.value(QStringLiteral("name")).toString();
        const QJsonObject args = call.value(QStringLiteral("arguments")).toObject();
        ++m_toolCalls;
        emit toolStep(m_area, name, "running");

        QString mcpName = name;
        QJsonObject payload = args;
        if (name == QLatin1String("inspect_object")) {
            // 虚拟工具：接口只有「看」的语义，脚本由后端固定拼，名字不安全就直接拒
            const QString objectName = args.value(QStringLiteral("name")).toString();
            mcpName = QStringLiteral("execute_blender_code");
            payload = QJsonObject{
                {"code", isSafeObjectName(objectName)
                             ? inspectObjectScript(objectName)
                             : QStringLiteral("print('{\"error\":\"unsafe-object-name\"}')")}};
        } else if (!kToolWhitelist.contains(name)) {
            // 模型硬编了一个白名单外的工具：本地拒绝，不转发给 MCP
            emit toolStep(m_area, name, "error");
            m_messages.append(QJsonObject{{"role", "tool"},
                                          {"tool_call_id", id},
                                          {"content", QJsonArray{QJsonObject{
                                              {"type", "text"},
                                              {"text", QStringLiteral("ERROR: 讨论模式只允许 "
                                                                      "get_scene_info / look / "
                                                                      "inspect_object")}}}}});
            if (--m_inFlight == 0) {
                m_stream.clear();
                streamOnce();
            }
            continue;
        }

        m_mcp->callTool(mcpName, payload, [this, id, name](const QJsonValue& result,
                                                           const QJsonObject& error) {
            emit toolStep(m_area, name, error.isEmpty() ? "done" : "error");
            QStringList images;
            m_messages.append(toolResultMessage(id, result, error, &images));
            for (const QString& dataUrl : images)
                emit delta(m_area, QString()); // 见 Step 3b：截图改由 imageReceived 信号外送
            if (--m_inFlight == 0) {
                m_stream.clear();
                streamOnce(); // 结果全部回喂后继续同一轮讨论
            }
        });
    }
}
```
